# ============================================================
# ASSOKIT - robots.txt
# Logiciel de gestion d'association loi 1901 et TPE
# https://assokit.fr
# ============================================================
#
# RÈGLE FONDAMENTALE : un robot n'obéit QU'AU SEUL groupe le plus
# spécifique qui correspond à son user-agent. Il n'hérite JAMAIS du
# groupe « * ». Autrement dit, tout groupe nommé doit répéter les
# interdictions, sinon ce robot a accès à tout le site.
#
# C'est pourquoi les groupes ci-dessous reprennent la liste complète
# au lieu de se contenter d'une directive Crawl-delay.
#
# Les paramètres d'URL (utm_, fbclid, gclid, ref=) ne sont volontairement
# PAS bloqués : chaque page publique déclare déjà sa balise canonical, et
# Google doit pouvoir la lire pour regrouper les variantes. Les bloquer
# l'empêchait de crawler, donc de canoniser — d'où les alertes
# « Bloquée par le fichier robots.txt » sur les liens de campagne.


# === TOUS LES ROBOTS (Google, Bing, et les robots IA) ===
# Les robots IA (GPTBot, ClaudeBot, PerplexityBot…) relèvent de ce groupe :
# ils accèdent donc à tout le site public, et à rien de la zone privée.
User-agent: *

# Pages privées (zone connectée)
Disallow: /dashboard
Disallow: /mon-asso-*
Disallow: /admin-*
Disallow: /super-admin
Disallow: /fondateur-*
Disallow: /projet/
Disallow: /projets
Disallow: /nouveau-projet
Disallow: /modifier-projet
Disallow: /supprimer-projet
Disallow: /action-projet
Disallow: /nouveau-dossier
Disallow: /supprimer-dossier
Disallow: /archives
Disallow: /prospection
Disallow: /mon-asso-qr
# Pages de collecte : on y arrive en scannant un QR, pas par Google
Disallow: /qr/

# Endpoints techniques
Disallow: /api/
Disallow: /cron/
Disallow: /cron-*
Disallow: /webhooks/
Disallow: /stripe-*
Disallow: /webhook-*

# Fichiers techniques
Disallow: /vendor/
Disallow: /config.php
Disallow: /*.log$
Disallow: /*.bak$
Disallow: /*.sql$
Disallow: /tmp/

# Recherche interne (résultats infinis, aucune valeur d'indexation)
Disallow: /blog?q=

# Pages d'authentification / compte
Disallow: /reset-password
Disallow: /mot-de-passe-oublie
Disallow: /definir-mot-de-passe
Disallow: /verify-email
Disallow: /verifier-email
Disallow: /connexion
Disallow: /signup
Disallow: /logout
Disallow: /deconnexion

# Cartes de visite numériques : coordonnées personnelles, rien à indexer
Disallow: /carte/

# Ressources nécessaires au rendu (à ne jamais bloquer)
Allow: /assets/
Allow: /asso/
